# 1. EoD refresh — pulls today's daily bar for tier-1 tickers (NIFTY 200 by default).
#    Standard production-cadence operation.
trip-ema-etl run-eod --exchange NSE

# 2. Intraday refresh — pulls recent bars including today's intraday data.
#    Set `intraday_lookback_days` in config (default 5).
trip-ema-etl run-intraday --exchange NSE

# 3. Backfill — for a date range, useful when seeding or after extended downtime.
trip-ema-etl run-backfill --exchange NSE --start 2026-05-01 --end 2026-05-19


OR via...

# Trigger an EoD run via HTTP — returns 202 Accepted, runs async on the etl_executor
curl -X POST \
  -H "Authorization: Bearer $TRIP_EMA_ADMIN_TOKEN" \
  http://localhost:8000/v1/admin/etl/run-eod \
  -H "Content-Type: application/json" \
  -d '{"exchange": "NSE"}'

# Check status afterward:
curl -H "Authorization: Bearer $TRIP_EMA_ADMIN_TOKEN" \
  http://localhost:8000/v1/admin/etl/runs | jq


OR via...

# Step 1: pull
trip-ema-etl run-eod --exchange NSE

# Step 2: force the orchestrator to fire a tick now (don't wait for the hourly)
curl -X POST -H "Authorization: Bearer $TRIP_EMA_ADMIN_TOKEN" \
  http://localhost:8000/v1/admin/orchestrator/NSE/run-now

# Step 3: watch the pulse stream for the result
curl -N http://localhost:8000/v1/pulse/stream
